\subsection{Install all Necessary Libraries for SNode.C}

\begin{frame}{Install all Necessary Libraries for SNode.C}
  \begin{itemize}
    \item Install all \textcolor{fh-ooe-red}{Hard Dependencies}:
    \begin{itemize}
      \item \texttt{\textcolor{fh-ooe-red}{libeasyloggingpp-dev}} for logging.
      \item \texttt{\textcolor{fh-ooe-red}{libssl-dev}} for SSL and TLS encrypted data exchange.
      \item \texttt{\textcolor{fh-ooe-red}{pkg-config}} package for cmake used to search for already installed libraries.
      \item \texttt{\textcolor{fh-ooe-red}{nlohmann-json3-dev}} for \textcolor{fh-ooe-red}{J}ava\textcolor{fh-ooe-red}{S}cript \textcolor{fh-ooe-red}{O}ptiont \textcolor{fh-ooe-red}{N}otation (\textcolor{fh-ooe-red}{JSON}) support.
    \end{itemize}
    \item Install all \textcolor{fh-ooe-red}{Soft Dependencies}:
    \begin{itemize}
      \item \texttt{\textcolor{fh-ooe-red}{clang-format}} to format the source code consistently.
      \item \texttt{\textcolor{fh-ooe-red}{cmake-format}} to format all CMakeLists.txt files consistently.
      \item \texttt{\textcolor{fh-ooe-red}{doxygen}} to create the documentation.
      \item \texttt{\textcolor{fh-ooe-red}{iwyu}} to prove that all required include files are explicitly included.
      \item \texttt{\textcolor{fh-ooe-red}{libbluetooth-dev}} for bluetooth transport.
      \item \texttt{\textcolor{fh-ooe-red}{libmagic-dev}} for mime-type-detection of files.
      \item \texttt{\textcolor{fh-ooe-red}{libmariadb-dev}} for database access.
    \end{itemize}
  \end{itemize}
\end{frame}
