\section{Install Debian Linux on Windows or MacOS (M1)}

\begin{frame}{Install Debian Linux on Windows or MacOS}
  \begin{enumerate}
    \item[Either] (best but complex) use \textcolor{fh-ooe-red}{Dual Boot} and install \textcolor{fh-ooe-red}{latest Debian} Linux \textcolor{fh-ooe-red}{nativly}.
    \begin{itemize}
      \item See e.g.: \textcolor{fh-ooe-red}{\href{https://getlabsdone.com/how-to-dual-boot-debian-11-and-windows-11/}{https://getlabsdone.com/how-to-dual-boot-debian-11-and-windows-11/}}.
    \end{itemize}
    \item[Or] (recommended) install \textcolor{fh-ooe-red}{VirtualBox} and \textcolor{fh-ooe-red}{latest Debian} Linux.
    \begin{itemize}
      \item See: \textcolor{fh-ooe-red}{\href{https://www.virtualbox.org/}{https://www.virtualbox.org/}}.
      \item Use a latest \textcolor{fh-ooe-red}{netinst-Image} from \textcolor{fh-ooe-red}{\href{https://www.debian.org/CD/netinst/}{https://www.debian.org/CD/netinst/}}
      \item Create and use a virtual harddrive with at least \textcolor{fh-ooe-red}{30 GB} free space.
      \item Assign at least \textcolor{fh-ooe-red}{4 GB} of \textcolor{fh-ooe-red}{RAM} to the \textcolor{fh-ooe-red}{Debian virtual machine}.
    \end{itemize}
    \item[Or] (not recommended) install Debian in Windows Subsystem for Linux (Windows 11 required).
    \begin{itemize}
      \item See: \textcolor{fh-ooe-red}{\href{https://wiki.debian.org/InstallingDebianOn/Microsoft/Windows/SubsystemForLinux}{https://wiki.debian.org/InstallingDebianOn/Microsoft/Windows/SubsystemForLinux}}.
    \end{itemize}
    \item[Or] (best for \textcolor{fh-ooe-red}{MacOS} on M1) install Debian Linux using \textcolor{fh-ooe-red}{Parallels}.
    \begin{itemize}
      \item See: \textcolor{fh-ooe-red}{\href{https://kb.parallels.com/128445}{https://kb.parallels.com/128445}}.
    \end{itemize}
    \item[Or] (recommended for \textcolor{fh-ooe-red}{MacOS} on Intel or M1) install \textcolor{fh-ooe-red}{VirtualBox} and latest Debian Linux.
    \begin{itemize}
      \item See installing \textcolor{fh-ooe-red}{VirtualBox} above.
    \end{itemize}
  \end{enumerate}
\end{frame}
