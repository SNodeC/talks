\section{Learning Objectives}

\begin{frame}{Learning Objectives}
\begin{itemize}
    \item \alert{Define and critically reflect} on the term “Internet of Things” as well as align and position Internet Of Things technology in comparison to cloud and mobile computing
    \item Equip most \alert{physical objects} with \alert{networked sensors}
    \item Equip and interact with most \alert{physical objects} with \alert{actors}
    \item \alert{Write code} to set-up systems to \alert{collect}, \alert{visualize}, \alert{analyze}, and \alert{act on data from the physical world}
    \item \alert{Design} and \alert{implement various IoT} systems made up of sensors, wireless network connection, and actuators
    \item \alert{Build} and \alert{test} a complete \alert{working IoT system} as well as design and implement respective simulator components to speed up testing
    \item \alert{Present and defend} the design and implementation of several \alert{Internet Of Things Systems} in a portfolio and public presentations
\end{itemize}
\end{frame}
