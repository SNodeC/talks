\part[nosectionpage]{What You Can Expect in Hagenberg}

\begin{frame}{What You Can Expect During the Three Weeks}
\begin{itemize}
    \item Work in an \textbf{international team} on one concrete stakeholder-driven project.
    \item Build with \textbf{real low-voltage hardware}: controllers, sensors, actuators and energy-related components.
    \item Connect distributed nodes through \textbf{LoRaWAN and The Things Network}.
    \item Move data through \textbf{MQTT}, persist it and make it visible in web applications.
    \item Integrate sensing, communication, backend logic and -- where useful -- \textbf{actuation and control}.
    \item Create data and system behaviour that later modules can use for \textbf{AI and Data Science}.
    \item End with a \textbf{working end-to-end demonstration}.
\end{itemize}
\begin{alertblock}{Three Intense Weeks}
Expect research, decisions, building, programming, integration, debugging, teamwork -- and visible progress every day.
\end{alertblock}
\end{frame}
