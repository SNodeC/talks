\begin{frame}{SolarScout -- Prototype Scope \& Required Resources}
\small
\begin{columns}[T,onlytextwidth]
    \begin{column}{0.52\textwidth}
        \textbf{Hardware}
        \begin{itemize}
            \item Portable ESP32-class LoRaWAN nodes
            \item Small reference PV cell or module
            \item Irradiance/light sensing and voltage/current measurement
            \item Ambient and panel-temperature sensors
            \item GNSS receiver
            \item IMU/inclinometer; optional magnetometer
            \item Adjustable mounting or small tripod
        \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
        \textbf{Software \& Infrastructure}
        \begin{itemize}
            \item LoRaWAN gateway and The Things Network
            \item MQTT / MQTTSuite backend
            \item Database with time and location data
            \item Web/map frontend for spatial comparison
        \end{itemize}
    \end{column}
\end{columns}

\begin{block}{Final Demonstration}
Measurements from different positions and configurations identify and compare locations with high and low solar potential.
\end{block}
\end{frame}
