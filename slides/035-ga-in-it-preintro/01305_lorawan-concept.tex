\begin{frame}{How Can IoT and LoRaWAN Support the Research?}{The Original Water-Monitoring Concept}
\begin{columns}[T,onlytextwidth]
\column{.48\textwidth}
\begin{block}{Autonomous Gateways}
\begin{itemize}
\item Place gateways around Hartbeespoort Dam.
\item Use GPS to identify their exact location.
\item Power them with solar panels and rechargeable batteries.
\item Connect them to The Things Network.
\end{itemize}
\end{block}
\column{.48\textwidth}
\begin{block}{Floating Measurement Nodes}
\begin{itemize}
\item Measure water quality and nutrient levels.
\item Add GPS location to the measurements.
\item Use solar panels and rechargeable batteries.
\item Send data via LoRaWAN gateways to The Things Network.
\end{itemize}
\end{block}
\end{columns}
\begin{alertblock}{A Web Interface Connects the System}
Manage the deployed devices and acquire their measurement data through a web application.
\end{alertblock}
\end{frame}
