\begin{frame}{Reconfigure the IoT system to use \texttt{mqttbroker}}
\begin{exampleblock}{Tasks (group): Port IoT Scenarious to \texttt{mqttbroker}}
  \begin{enumerate}
    \item \alert{Duplicate all} your \alert{arduino sketches} and \alert{iotempower configurations}.
    \begin{description}[$\Rightarrow$]
      \item[$\Rightarrow$] Do this \alert{systematically} to \alert{not get confused}.
    \end{description}
    \item \alert{Duplicate all} your \alert{Node-Red flows}.
    \begin{description}[$\Rightarrow$]
      \item[$\Rightarrow$] Do this \alert{systematically} to \alert{not get confused}.
    \end{description}
    \item Reconfigure selected \alert{IoT scenarious} to utilize \alert{\texttt{mqttbroker}} instead of \texttt{mosquitto}.
    \begin{itemize}
    \item Reconfigure selected \alert{arduino sketches} and \alert{iotempower} configurations to contact the \alert{\texttt{mqttbroker}} at IP address \alert{\texttt{192.168.12.254}} instead of \texttt{192.168.12.1}.
    \item Reconfigure selected \alert{Node-Red flows} to contact the \alert{\texttt{mqttbroker}} at IP address \alert{\texttt{192.168.12.254}} instead of \texttt{192.168.12.1}.
    \end{itemize}
    \begin{description}[$\Rightarrow$ Hint:]
      \item[$\Rightarrow$ Hint:] First port only one \alert{simple sketch} or \alert{simple iotempower configuration} and the \alert{corresponding Node-Red flow} to \alert{test the infrastructure}.
      \item[$\Rightarrow$ Hint:]  The \alert{ESP32 controler}, desireably, should utilize the \alert{router WiFi} instead of the RasPI one.
    \end{description}
  \end{enumerate}
  \end{exampleblock}
\end{frame}
