\section{Applikationsprotokolle als Sub-Frameworks}

\begin{frame}{Applikationsprotokolle als Sub-Frameworks}{Aktueller Stand}
  \begin{description}[WebSocket (WS):]
    \item [HTTP/1.1:] Server- und Clientseitig inkl. virtual Hosts support.
    \item [HTTP/Express:] Serverseitiges API äquivalent zu Node.JS/Express
    \item [WebSocket (WS):] Server- und Clientseitig
    \item [MQTT:] Server- und Clientseitig (MQTT v3.1.1, v5.0 in Planung)
    \item [MQTT via WS:] Server- und Clientseitig (was MQTT kann, kann MQTT via WS automatisch auch)
  \end{description}
\end{frame}
