\subsection{The RFCOMM Server Application}

\begin{frame}[fragile]{The RFCOMM Server Application}
  \begin{cppcode}[autogobble=true, caption={RFCOMM server application.},captionpos=b]
    #include "EchoServerContextFactory.h"
    #include <core/SNodeC.h>
    #include <iostream>
    #include <net/rc/stream/legacy/SocketServer.h>
    #include <cstring>
    #include <string>

    int main(int argc, char* argv[]) {
      core::SNodeC::init(argc, argv);

      using EchoServer = net::rc::stream::legacy::SocketServer<EchoServerContextFactory>;
      using SocketAddress = EchoServer::SocketAddress;

      EchoServer echoServer;

      echoServer.listen(1, [](const core::ProgressLog& progressLog) -> void {
        progressLog.logProgress();
      });

      return core::SNodeC::start();
    }
  \end{cppcode}
\end{frame}
