\subsection{Hauptprogramm des IPv4-Servers}

\begin{frame}[fragile]{Hauptprogramm des IPv4-Servers}
  \begin{cppcode}[autogobble=true]
    #include "EchoServerContextFactory.h"
    #include <core/SNodeC.h>
    #include <iostream>
    #include <net/in/stream/legacy/SocketServer.h>
    #include <cstring>
    #include <string>

    int main(int argc, char* argv[]) {
      core::SNodeC::init(argc, argv);

      using EchoServer = net::in::stream::legacy::SocketServer<EchoServerContextFactory>;
      using SocketAddress = EchoServer::SocketAddress;

      EchoServer echoServer;

      echoServer.listen(8001, [](const core::ProgressLog& progressLog) -> void {
        progressLog.logProgress();
      });

      return core::SNodeC::start();
    }
  \end{cppcode}
\end{frame}
