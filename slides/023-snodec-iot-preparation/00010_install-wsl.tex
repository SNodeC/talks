\section{Install Debian Linux on Windows or MacOS (M1)}
\withsectionpage

\begin{frame}{Install Debian Linux on Windows or MacOS}
  \begin{enumerate}
    \item[Either] (best but complex) use \alert{Dual Boot} and install \alert{latest Debian} Linux \alert{nativly}.
    \begin{itemize}
      \item See e.g.: \alert{\href{https://getlabsdone.com/how-to-dual-boot-debian-11-and-windows-11/}{https://getlabsdone.com/how-to-dual-boot-debian-11-and-windows-11/}}.
    \end{itemize}
    \item[Or] (recommended) install \alert{VirtualBox} and \alert{latest Debian} Linux.
    \begin{itemize}
      \item See: \alert{\href{https://www.virtualbox.org/}{https://www.virtualbox.org/}}.
      \item Use a latest \alert{netinst-Image} from \alert{\href{https://www.debian.org/CD/netinst/}{https://www.debian.org/CD/netinst/}}
      \item Create and use a virtual harddrive with at least \alert{30 GB} free space.
      \item Assign at least \alert{4 GB} of \alert{RAM} to the \alert{Debian virtual machine}.
    \end{itemize}
    \item[Or] (not recommended) install Debian in Windows Subsystem for Linux (Windows 11 required).
    \begin{itemize}
      \item See: \alert{\href{https://wiki.debian.org/InstallingDebianOn/Microsoft/Windows/SubsystemForLinux}{https://wiki.debian.org/InstallingDebianOn/Microsoft/Windows/SubsystemForLinux}}.
    \end{itemize}
    \item[Or] (best for \alert{MacOS} on M1) install Debian Linux using \alert{Parallels}.
    \begin{itemize}
      \item See: \alert{\href{https://kb.parallels.com/128445}{https://kb.parallels.com/128445}}.
    \end{itemize}
    \item[Or] (recommended for \alert{MacOS} on Intel or M1) install \alert{VirtualBox} and latest Debian Linux.
    \begin{itemize}
      \item See installing \alert{VirtualBox} above.
    \end{itemize}
  \end{enumerate}
\end{frame}
