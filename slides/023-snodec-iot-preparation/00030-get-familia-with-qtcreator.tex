\section{Become Familiar with QT-Creator}

\begin{frame}[fragile]{Become Familiar with QT-Creator}
  \begin{itemize}
    \item Create a first C++ project in QT-Creator using \texttt{cmake} as {\em Build System}.
    \item Try to implement and compile a \textcolor{fh-ooe-red}{Hello, World!} application.
    \begin{cppcode}[gobble=6]
      #include <iostream>
      #include <cstdlib>

      int main(int argc, char* argv[]) {
        std::cout << "Hello, World!" << std::endl;

        return EXIT_SUCCESS;
      }
    \end{cppcode}
    \item Questions:
    \begin{enumerate}
      \item How is this application deployed (installed) for system-wide use? Search for \textcolor{fh-ooe-red}{cmake install application}
      \item How are the in-build-tree and system-wide-installed executables started?
    \end{enumerate}
    \item Modify \textcolor{fh-ooe-red}{CMakeLists.txt} so, that installation of the application for system-wide use is possible.
    \item Install the application and execute the in-build-tree and system-wide-installed executable.
  \end{itemize}
\end{frame}
