\part[nopartpage]{Overview}

\begin{frame}{Overview}
    The \alert{MQTTSuite} provides a full \alert{set of applications} to create complext and versatile \alert{IoT scenarios}.
    \begin{itemize}
        \item While \alert{fully functional} and \alert{ready for production}, the applications should be \alert{treeted as templates}.
        \begin{itemize}
            \item \alert{Modifikations} and \alert{adjustments} can be made easily by \alert{adding/removing} additional \alert{server or client} instances to support \alert{alternative network and connection} types.
    \end{itemize}
    \end{itemize}
    \begin{description}[MQTTIntegrator]
        \item [MQTTBroker]
        \begin{itemize}
            \item Fully supports version \href{https://docs.oasis-open.org/mqtt/mqtt/v3.1.1/mqtt-v3.1.1.html}{\textcolor{fh-ooe-red}{3.1.1 of the MQTT Standard} (10. Dec. 2015)}
            \item Message \alert{retaining}
            \item Session \alert{management} and \alert{hibernating}
            \item Can \alert{act as} an MQTT \alert{integrator}
            \item Supports \alert{message-loop-prevention} like and compatible with \href{https://mosquitto.org/}{\textcolor{fh-ooe-red}{mosquitto}}
        \end{itemize}
        \item [MQTTIntegrator]
        \begin{itemize}
            \item Connects to a MQTT-Broker as MQTT-Client
            \item \alert{Subscribes} to selected \alert{topics}
            \item \alert{Messages} received via \alert{topics} are \alert{mapped} to \alert{result messages and topics}
            \item Topic/message \alert{mapping described} and controlled \alert{via JSON}
        \end{itemize}
        \item [MQTTBridge]
        \begin{itemize}
            \item Connects to \alert{multiple} MQTT-\alert{Brokers}
            \item Subscribes to selected \alert{topics individually} for each MQTT-\alert{Broker}
            \item \alert{Bridges messages} of subscribed \alert{topics} between Brokers
            \item Supports \alert{message-loop-prevention} like and compatible with \href{https://mosquitto.org/}{\textcolor{fh-ooe-red}{mosquitto}}
            \item Contacted Brokers and subscribed topics controlled vial \alert{JSON}
        \end{itemize}
    \end{description}
\end{frame}
