\part{Configuration of IoT Scenarious using the \texttt{mqttintegrator}}

\section[nosectionpage]{The JSON Structure of the Mapping File used by \texttt{mpttintegrator}}
\begin{frame}[fragile]{The JSON Structure of the Mapping File used by \texttt{mpttintegrator}}
  The big Structure of the JSON mapping description consists of three objects:
  \begin{description}[connection]
    \item[discover\_prefix] The prefix topic-level of the discover topics of an application.
    \item[connection] This object describes options for the MQTT connection.
    \item[mappings] This object describes the mappings of incoming topics|messages to outgoing topics|messages.
  \end{description}

  \begin{jsoncode}[gobble=4]
    {
        "discover_prefix" : "iotempower",  // discover prefix of the application
        "connection" : {                   // connection options
            "keep_alive" : 60,             // [60]    duration between two ping packets
            "client_id" : "Client",        // [""]    logical name of the client
            "clean_session" : true,        // [true]  start with clean session or pick up retained one
            "will_topic" : "will/topic",   // [""]    last will topic
            "will_message" : "Last Will",  // [""]    last will message
            "will_qos": 0,                 // [0]     last will qos
            "will_retain" : false,         // [false] retain last will or not
            "username" : "Username",       // [""]    username
            "password" : "Password"        // [""]    password
        },
        "mappings" : {                     // mapping descriptions of templates|messages mapping
            ...
        }
    }
  \end{jsoncode}
\end{frame}
