\section{Topics, QoS, Retained Messages und Last Will}

\begin{frame}{Topics, QoS, Retained Messages und Last Will}
  \begin{itemize}
\item \textbf{Topics sind hierarchisch}:
	\begin{itemize}
\item \texttt{home/kitchen/temp},
\item \texttt{factory/line1/motor/rpm}
	\end{itemize}
\item \textbf{Wildcards} (common):
	\begin{itemize}
\item \texttt{+} = one level (\texttt{home/+/temp})
\item \texttt{\#} = many levels (\texttt{home/\#})
	\end{itemize}
\item \textbf{QoS (Quality of Service)} = delivery guarantee level:

  \vspace{0.5ex}
  \begin{tabular}{lll}
    \toprule
    \textbf{QoS} & \textbf{Name} & \textbf{Meaning} \\
    \midrule
    0 & At most once  & Fastest, no retries; message may be lost \\
    1 & At least once & Retries until ACK; duplicates possible \\
    2 & Exactly once  & Highest reliability; more overhead/handshakes \\
    \bottomrule
  \end{tabular}

  \vspace{0.5ex}
\item \textbf{Retained messages} (often used)
  \begin{itemize}
\item Broker stores last value per topic
  \end{itemize}

  \vspace{0.5ex}
\item \textbf{Last Will}
  \begin{itemize}
\item Broker publishes a “client died” message if a client disconnects unexpectedly
  \end{itemize}
  \end{itemize}
\end{frame}
