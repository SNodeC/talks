\part{Message Queuing Telemetry Transport (MQTT)}

\begin{frame}{Was ist MQTT?}

  \centering\includegraphics[width=0.6\linewidth]{images/MQTT}%
  \vspace{2em}
  \begin{itemize}
\item \alert{MQTT} ist ein leichtgewichtiges \alert{publish/subscribe} messaging protocol.
\item Ausgelegt für \alert{small devices} and \alert{unzuverlässige/niedrigbandbreitige} networks (IoT).
\item Läuft typischerweise über \alert{TCP/IP}; Clients verbinden sich mit einem \alert{broker} (server).
\item Nachrichten werden an \alert{topics} (like channel names), nicht direkt an andere Clients.
  \end{itemize}

  \vspace{1ex}
  \footnotesize
  Kernidee: \alert{Entkopplung} von Datenproduzenten (Sensoren) und Datenkonsumenten (Apps/Dashboards).
\end{frame}
