\begin{frame}{Beispielhafte Anwendungsfälle von LoRa und LoRaWAN}
    \begin{description}[Supply Chain and Logistics]
        \item[Smart Cities]
            \begin{itemize}
\item Intelligente Straßenbeleuchtung
\item Abfallwirtschaft (z.\,B. Füllstandsmessung)
\item Parksensorik
            \end{itemize}
        \item[Landwirtschaft]
            \begin{itemize}
\item Bodenfeuchte-Monitoring
\item Bewässerungssteuerung
\item Tracking von Nutztieren
            \end{itemize}
        \item[Industrielles IoT (IIoT)]
            \begin{itemize}
\item Vorausschauende Wartung (Predictive Maintenance)
\item Asset-Tracking
\item Fernüberwachung von Anlagen
            \end{itemize}
        \item[Umweltmonitoring]
            \begin{itemize}
\item Überwachung von Luft- und Wasserqualität
\item Hochwasser-Früherkennungssysteme
\item Wildtier-Monitoring
            \end{itemize}
        \item[Lieferkette und Logistik]
            \begin{itemize}
\item Echtzeit-Tracking von Sendungen während des Transports
\item Kühlketten-Monitoring: Temperatur- und Feuchteüberwachung für sensible Güter
\item Lagerbestands-Tracking
            \end{itemize}
    \end{description}
\end{frame}
