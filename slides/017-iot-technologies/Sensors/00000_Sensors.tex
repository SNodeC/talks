% ============================================================
\section[withsectionpage]{Totaly Disolved Solids (TDS) Sensor}
\begin{frame}{TDS Sensor (B08KXRHK7H): What it Measures}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.62\textwidth}
    \begin{itemize}
      \item \textbf{TDS = Total Dissolved Solids} (ppm) \\
            $\rightarrow$ proxy via \textbf{electrical conductivity}
      \item Typical use: water quality monitoring (freshwater / hydroponics / labs)
      \item This board provides \textbf{analog voltage output} for a microcontroller
      \item Input supply range: works from \textbf{3.3\,V to 5\,V-class} supplies
      \item Practical note: \textbf{temperature compensation} improves accuracy
    \end{itemize}
  \end{column}
  \begin{column}{0.38\textwidth}
    \centering
    \includegraphics[width=0.8\linewidth]{figures/TDS}\\
    \scriptsize Example: CQRobot TDS module + probe
  \end{column}
\end{columns}
\end{frame}

\begin{frame}{TDS sensor + ESP32: Wiring \& Usage}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.62\textwidth}
    \begin{itemize}
      \item \textbf{Power:} connect to 3.3\,V (or 5\,V if required) + GND
      \item \textbf{Signal:} analog OUT $\rightarrow$ \textbf{ESP32 ADC1 pin}
            \begin{itemize}
              \item Use \textbf{ADC1} (ADC2 conflicts with Wi-Fi on ESP32)
              \item Oversample + average (e.g., 50--200 samples)
            \end{itemize}
      \item \textbf{Conversion:} voltage $\rightarrow$ conductivity/TDS using vendor formula
      \item Optional: use an \textbf{external ADC (ADS1115)} for better resolution
    \end{itemize}
    \vspace{0.3em}
    \footnotesize
    Tip: keep the probe clean; avoid bubbles; measure in flowing water for stability.
  \end{column}
  \begin{column}{0.38\textwidth}
    \centering
%    \includegraphics[width=\linewidth]{figures/tds_wiring_ads1115.png}\\
    \scriptsize Example wiring with external ADC
  \end{column}
\end{columns}
\end{frame}

\section[withsectionpage]{Turbidity Sensor}
\begin{frame}{Turbidity Sensor (B07SLRRX38 / TSW-20M): Overview}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.62\textwidth}
    \begin{itemize}
      \item \textbf{Turbidity} = “cloudiness” from particles in water (optical method)
      \item Used for: wastewater, rivers/lakes, washing/dishwasher monitoring
      \item Module typically provides:
            \begin{itemize}
              \item \textbf{Analog output} (continuous turbidity signal)
              \item \textbf{Digital output} (threshold comparator)
            \end{itemize}
      \item Often powered from \textbf{5\,V} (sensor + conditioning board)
    \end{itemize}
  \end{column}
  \begin{column}{0.38\textwidth}
    \centering
    \includegraphics[width=0.8\linewidth]{figures/Turbidity}\\
    \scriptsize TSW-20M style turbidity sensor + board
  \end{column}
\end{columns}
\end{frame}

\begin{frame}{Turbidity + ESP32: Practical Integration}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.62\textwidth}
    \begin{itemize}
      \item \textbf{Power:} 5\,V + GND (share GND with ESP32)
      \item \textbf{Analog OUT:}
            \begin{itemize}
              \item If output can exceed 3.3\,V: use a \textbf{voltage divider} into ADC1
              \item Calibrate: “clear water” vs “known cloudy water” samples
            \end{itemize}
      \item \textbf{Digital OUT:} use for simple \textbf{clear/cloudy} threshold trigger
      \item Maintenance: keep optical window clean (biofilm strongly affects readings)
    \end{itemize}
  \end{column}
  \begin{column}{0.38\textwidth}
    \centering
%    \includegraphics[width=\linewidth]{figures/turbidity_board_pins.jpg}\\
    \scriptsize Typical pins: V, G, A(analog), D(digital)
  \end{column}
\end{columns}
\end{frame}

\section[withsectionpage]{pH Sensor Kit}
\begin{frame}{pH Sensor Kit (B0B7NN3TSC / B08T645TYQ): Overview}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.5\textwidth}
    \begin{itemize}
      \item \textbf{pH} describes acidity/alkalinity: scale \textbf{0--14}
      \item A glass \textbf{pH electrode} outputs a tiny voltage (very high impedance)
      \item The interface board:
            \begin{itemize}
              \item powers/conditions the electrode signal
              \item provides an \textbf{analog voltage output} for a microcontroller ADC
              \item often uses a \textbf{BNC connector} for the probe
            \end{itemize}
      \item Needs \textbf{calibration} with buffer solutions (pH 4/7/10)
    \end{itemize}
  \end{column}
  \begin{column}{0.5\textwidth}
    \centering
    \includegraphics[width=0.49\linewidth]{figures/PH-Interface+Temperature} \includegraphics[width=0.49\linewidth]{figures/PH-Electrode}\\
    \scriptsize Typical pH-4502C style board + probe
  \end{column}
\end{columns}
\end{frame}

\begin{frame}{pH + ESP32: Wiring \& Measurement Workflow}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.62\textwidth}
    \begin{itemize}
      \item \textbf{Power:} many boards expect \textbf{5\,V} (check yours) + GND
      \item \textbf{Analog OUT $\rightarrow$ ESP32 ADC1}
            \begin{itemize}
              \item If OUT may exceed 3.3\,V: \textbf{divider} into ESP32
              \item Use averaging + stable sampling time (electrodes are slow)
            \end{itemize}
      \item \textbf{Calibration:}
            \begin{itemize}
              \item measure buffers, adjust offset/slope (often 2 trim pots)
              \item store calibration constants in flash/NVS
            \end{itemize}
      \item Optional: temperature sensor for \textbf{temperature compensation}
    \end{itemize}
  \end{column}
  \begin{column}{0.38\textwidth}
    \centering

    \scriptsize BNC + analog output to ADC
  \end{column}
\end{columns}
\end{frame}

\section[withsectionpage]{Global Positioning System (GPS)}
\begin{frame}{GT-U7 GPS Module (B0CNGWV7MN): What it Gives You}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.62\textwidth}
    \begin{itemize}
      \item Provides \textbf{position + time} from GNSS satellites
      \item Outputs standard \textbf{NMEA sentences} over \textbf{UART serial}
      \item Default baud rate is commonly \textbf{9600 bps}
      \item Needs good antenna placement (near window / outdoors for fastest lock)
      \item Typical outputs:
            latitude, longitude, altitude, speed, UTC time, satellites in view
    \end{itemize}
  \end{column}
  \begin{column}{0.38\textwidth}
    \centering
    \includegraphics[width=0.7\linewidth]{figures/GPS-GT-U7}\\
    \scriptsize GT-U7 module (patch antenna + UART pins)
  \end{column}
\end{columns}
\end{frame}

\begin{frame}{GT-U7 + ESP32: Wiring \& Decoding}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.62\textwidth}
    \begin{itemize}
      \item \textbf{Power:} module supply (often 3.3--5\,V) + GND
      \item \textbf{UART wiring:}
            \begin{itemize}
              \item GPS \textbf{TX} $\rightarrow$ ESP32 \textbf{RX} (e.g., UART2)
              \item GPS \textbf{RX} $\leftarrow$ ESP32 \textbf{TX} (optional)
              \item Check logic levels; level-shift if your module TX is 5\,V
            \end{itemize}
      \item \textbf{Software:} parse NMEA with TinyGPSPlus (Arduino) or ESP-IDF parser
      \item Use-case idea: add GPS position to MQTT telemetry (asset tracking / buoys)
    \end{itemize}
  \end{column}

\section[withsectionpage]{LoRa Interface}
\begin{frame}{LoRa SX1262: What it is}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.62\textwidth}
    \begin{itemize}
      \item \textbf{SX1262 = sub-GHz LoRa transceiver} (radio chip, not a MCU)
      \item Supports \textbf{LoRa modulation} for long range + \textbf{(G)FSK} for legacy modes
      \item Key feature: \textbf{up to +22\,dBm TX power} (vs SX1261 typically +15\,dBm)
      \item Optimized for \textbf{low power} LPWAN applications
      \item Typical frequency bands: \textbf{EU868}, \textbf{US915}, others (depends on module + antenna)
      \item Used in: sensors, trackers, buoys, remote meters, industrial telemetry
    \end{itemize}
    \vspace{0.3em}
    \footnotesize
    Mental model: ESP32 does the application + protocol; SX1262 does the RF link.
  \end{column}

\begin{frame}{SX1262 + ESP32: Wiring \& Software Pattern}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.55\textwidth}
    \begin{itemize}
      \item Interface is \textbf{SPI} + a few control/status pins:
        \begin{itemize}
          \item \textbf{NSS/CS}, \textbf{SCK}, \textbf{MOSI}, \textbf{MISO}
          \item \textbf{DIO1} (IRQ line, signals TX done / RX done / timeouts)
          \item \textbf{BUSY} (chip is busy, host must wait before SPI command)
          \item \textbf{NRST} (reset)
        \end{itemize}
      \item Antenna: keep it \textbf{50\,\(\Omega\)} and use the correct band (EU868 antenna for EU868)
      \item Software options:
        \begin{itemize}
          \item \textbf{Radio-only} (P2P LoRa): RadioLib, SX126x driver
          \item \textbf{LoRaWAN} (TTN): LoRaWAN stack + SX1262 radio driver
        \end{itemize}
      \item Typical RadioLib constructor pattern: \texttt{Module(NSS, DIO1, RST, BUSY)}
    \end{itemize}
  \end{column}
