\section[nosectionpage]{Sensor für Total Dissolved Solids (TDS)}
\begin{frame}{TDS-Sensor (B08KXRHK7H): Was er misst}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.62\textwidth}
    \begin{itemize}
\item \alert{TDS = Total Dissolved Solids} (ppm) \\
            $\rightarrow$ Proxy über die \alert{elektrische Leitfähigkeit}
\item Typischer Einsatz: Monitoring der Wasserqualität (Süßwasser / Hydroponik / Labore)
\item Dieses Board liefert \alert{analoge Spannungsausgänge} für einen Mikrocontroller
\item Versorgungsspannungsbereich: funktioniert mit \alert{\SIrange{3.3}{5}{\volt}}-Versorgung
\item Praktischer Hinweis: \alert{Temperaturkompensation} erhöht die Genauigkeit
    \end{itemize}
  \end{column}
  \begin{column}{0.38\textwidth}
    \centering
    \includegraphics[width=0.8\linewidth]{figures/TDS}\\
    \scriptsize Beispiel: CQRobot TDS-Modul + Sonde
  \end{column}
\end{columns}
\end{frame}

\begin{frame}{Turbidity Sensor (B07SLRRX38 / TSW-20M): Überblick}

            \begin{itemize}
\item \alert{Analogausgang} (kontinuierliches Trübungs-Signal)

            \end{itemize}
\item Häufig versorgt über \alert{\SI{5}{\volt}} (Sensor + Conditioning-Board)
    \end{itemize}

\section[nosectionpage]{pH Sensor Kit}
\begin{frame}{pH-Sensor-Kit (B0B7NN3TSC / B08T645TYQ): Überblick}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.5\textwidth}
    \begin{itemize}
\item \alert{pH} beschreibt den Säure-/Basengrad: Skala \alert{0--14}
\item Eine \alert{pH-Glaselektrode} liefert eine sehr kleine Spannung (sehr hohe Impedanz)
\item Das Interface-Board:
            \begin{itemize}
\item versorgt/konditioniert das Elektroden-Signal
\item liefert einen \alert{analogen Spannungsausgang} für den ADC eines Mikrocontrollers
\item verwendet häufig einen \alert{BNC-Stecker} für die Sonde
            \end{itemize}
\item Benötigt \alert{Kalibrierung} mit Pufferlösungen (pH 4/7/10)
    \end{itemize}
  \end{column}

    \scriptsize Typisches Board im pH-4502C-Stil + Sonde
  \end{column}
\end{columns}
\end{frame}

\section[nosectionpage]{Global Positioning System (GPS)}
\begin{frame}{GT-U7 GPS-Modul (B0CNGWV7MN): Was es liefert}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.62\textwidth}
    \begin{itemize}
\item Liefert \alert{Position + Zeit} von GNSS-Satelliten
\item Gibt standardisierte \alert{NMEA-Sätze} über \alert{UART (seriell)} aus
\item Die Standard-Baudrate ist häufig \alert{\SI{9600}{\bit\per\second}}
\item Benötigt eine gute Antennenplatzierung (nahe Fenster / im Freien für den schnellsten Fix)
\item Typische Ausgaben:
            Latitude, Longitude, Höhe, Geschwindigkeit, UTC-Zeit, sichtbare Satelliten
    \end{itemize}

\section[nosectionpage]{LoRa-Interface}
\begin{frame}{LoRa SX1262: Was es ist}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.62\textwidth}
    \begin{itemize}
\item \alert{SX1262 = Sub-GHz-LoRa-Transceiver} (Funkchip, keine MCU)
\item Unterstützt \alert{LoRa-Modulation} für große Reichweite sowie \alert{(G)FSK} für Legacy-Modi
\item Kernmerkmal: \alert{bis zu \SI{22}{\dBm} Sendeleistung} (SX1261 typischerweise \SI{15}{\dBm})
\item Optimiert für \alert{Low-Power}-LPWAN-Anwendungen
\item Typische Frequenzbänder: \alert{EU868}, \alert{US915}, weitere (abhängig von Modul + Antenne)
\item Einsatz in: Sensoren, Tracker, Bojen, Zähler/Fernablesung, industrielle Telemetrie
    \end{itemize}
    \vspace{0.3em}
    \footnotesize
    Mentales Modell: Der ESP32 übernimmt Applikation und Protokoll; der SX1262 realisiert die Funkstrecke.
  \end{column}

\begin{frame}{SX1262 + ESP32: Verdrahtung \& Software-Pattern}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.55\textwidth}
    \begin{itemize}
\item Schnittstelle: \alert{SPI} + einige Control-/Status-Pins:
        \begin{itemize}
\item \alert{NSS/CS}, \alert{SCK}, \alert{MOSI}, \alert{MISO}
\item \alert{DIO1} (IRQ-Leitung: signalisiert TX done / RX done / Timeouts)
\item \alert{BUSY} (Chip ist beschäftigt; Host muss vor dem SPI-Befehl warten)
\item \alert{NRST} (Reset)
         \end{itemize}
\item Antenne: auf \alert{\SI{50}{\ohm}} auslegen und das richtige Band verwenden (EU868-Antenne für EU868)
\item Software-Optionen:
         \begin{itemize}
\item \alert{Nur Radio} (P2P LoRa): RadioLib, SX126x-Treiber
\item \alert{LoRaWAN} (TTN): LoRaWAN-Stack + SX1262-Radio-Treiber
        \end{itemize}
\item Typisches RadioLib-Konstruktor-Pattern: \texttt{Module(NSS, DIO1, RST, BUSY)}
    \end{itemize}
  \end{column}
