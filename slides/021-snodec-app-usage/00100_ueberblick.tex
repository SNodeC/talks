\begin{frame}[fragile]{Configuration System Fundamentals}
  \begin{block}{\alert{SNode.C} offers three Options for Configuration}
    All of these options rely on the excellent \href{https://github.com/CLIUtils/CLI11}{\texttt{\alert{CLI11:Command line parser for C++11}}}:
    \begin{itemize}
      \item Command Line
      \item Configuration Files
      \item API for in-code configuration
    \end{itemize}
    Configuration Items exist for an
    \begin{itemize}
      \item \alert{application as such} and
      \item separately for \alert{every individual instance}
    \end{itemize}
    of an application.
  \end{block}
\end{frame}
