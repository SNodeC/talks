\subsection{IPv4 and IPv6: Sections \textcolor{fh-ooe-red}{\lstinline|local|} and \textcolor{fh-ooe-red}{\lstinline|remote|}}

\begin{frame}[fragile]{Sections \alert{\lstinline|local|} and \alert{\lstinline|remote|} of IPv4 and IPv6 Server and Client Instances}
The \alert{scope resolution operators} \textcolor{fh-ooe-red}{\lstinline|Local::|} and \textcolor{fh-ooe-red}{\lstinline|Remote::|} are used to \alert{resolve ambiguities} and to precisely define which \alert{side of a socket} is to be \alert{configured}.

\end{description}
\begin{footnotesize}
\begin{Verbatim}[commandchars=\\\{\},frame=leftline,rulecolor=\color{fh-ooe-red}]
\textcolor{purple}{auto}& \textcolor{blue}{\bf{}configObject} = \textcolor{blue}{ipServer}.\textcolor{grassgreen!60!black}{getConfig}();

\textcolor{blue}{configObject}.\textcolor{fh-ooe-red}{Local}::\textcolor{grassgreen!60!black}{setHost}(\textcolor{olive}{const} \textcolor{purple}{std::string}&);
\textcolor{blue}{configObject}.\textcolor{fh-ooe-red}{Local}::\textcolor{grassgreen!60!black}{setPort}(\textcolor{purple}{uint32_t});

\textcolor{blue}{configObject}.\textcolor{fh-ooe-red}{Local}::\textcolor{grassgreen!60!black}{setNumericReverse}(\textcolor{purple}{bool} = true);
