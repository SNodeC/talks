\part{Topics of second Part in IoT}
\section{Focus on Gateway and Cloud}
\subsection{Gateway}

\begin{frame}{Focus on Gateway and Cloud}
  \begin{description}
    \item[Gateway]\hfill\\
    \begin{itemize}
      \item Provide connection-endpoints for microcontroller
      \item Preprocessing and managing sensor- and actuator-data
      \item Provides IoT-system (application) logic
      \item Communication with cloud
      \item Security considerations
      \begin{itemize}
        \item User management
        \item Secure communication (SSL/TLS, https, certificates, \ldots)
        \item Cloud-To-Cloud accout linking (OAuth2, \ldots)
      \end{itemize}
      \item (Rest)-API for configuration and management
    \end{itemize}
  \end{description}
\end{frame}
