\part{Components of IoT}
\withpartpage

\begin{frame}{Components of IoT}
  \begin{columns}[onlytextwidth]
    \begin{column}[T]{0.5\textwidth}
      \begin{description}[Microcontroller]
        \item[Sensors]
        \begin{itemize}
          \item Temerature, humidity, pressure
          \item Brightness, acoustics
          \item Switches
          \item Chemicals
          \item \ldots
        \end{itemize}
        \item[Microcontroller]
        \begin{itemize}
          \item Controls sensors and actuators
          \item Preprocessing of sensor-data
          \item (Limited) networking capabilities
          \begin{itemize}
            \item WiFi, Bluetooth
            \item \ldots
          \end{itemize}
          \item Connected to Gateway
          \item \ldots
        \end{itemize}
      \end{description}
    \end{column}
    \begin{column}[T]{0.5\textwidth}
      \begin{description}[Actuators]
        \item[Actuators]
        \begin{itemize}
          \item Lamps
          \item Motors
          \item Locks
          \item Relays
          \item \ldots
        \end{itemize}
        \item[Gateway]
        \begin{itemize}
          \item Full network capabilities
          \item Collecting data from MC
          \item Sending commands to MC
          \item (Can) offer portal for whole IoT-system
          \item (Frequently) connected to cloud-system
          \item Responsible for security
          \item \ldots
        \end{itemize}
      \end{description}
    \end{column}
  \end{columns}
\end{frame}
