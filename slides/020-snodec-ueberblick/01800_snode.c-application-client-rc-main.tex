    #include "EchoClientContextFactory.h"
    #include <core/SNodeC.h>
    #include <iostream>
    #include <net/rc/stream/legacy/SocketClient.h>
    #include <cstring>
    #include <string>

    int main(int argc, char* argv[]) {
      core::SNodeC::init(argc, argv);

      using EchoClient = net::rc::stream::legacy::SocketClient<EchoClientContextFactory>;
      using SocketAddress = EchoClient::SocketAddress;

      EchoClient echoClient;

      echoClient.connect("10:3D:1C:AC:BA:9C", 1, [](const core::ProgressLog& progressLog) -> void {
        progressLog.logProgress();
      });

      return core::SNodeC::start();
    }
