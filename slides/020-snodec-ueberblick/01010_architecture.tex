\subsection{EventLoop, EventMultiplexer, EventQueue, Event and EventReceiver}

\begin{frame}{EventLoop, EventMultiplexer, EventQueue, Event and EventReceiver}
  \begin{center}
    \begin{tikzpicture}
      \umlclass[x=-3,fill=blue!20]{SNodeC}{}{}
      \umlclass[x=-0.5,fill=red!20]{EventLoop}{%
      }{
        tick(t : TimeVal) : int
      }
      \umlclass[x=3,y=0,type=abstract,fill=brown!20]{EventMultiplexer}{
      }{
        \umlvirt{multiplex () : void}
      }
      \umlclass[x=7,y=0]{EventQueue}{}{
        insert(event : Event*) : void\\
        execute() : void;
      }

      \umlclass[x=0.5,y=-2.5]{TimerEventPublisher}{}{
        insert(t : TimerEventReceiver)\\
        remove(t : TimerEventReceiver)\\
        publishActiveEvents() : void
      }
      \umlclass[x=5.5,y=-2.5,type=abstract,fill=brown!20]{DescriptorEventPublisher}{}{
        enable(d : DescriptorEventReceiver)\\
        disable(d : DescriptorEventReceiver)\\
        suspend(d : DescriptorEventReceiver)\\
        resume(d : DescriptorEventReceiver)\\
        \umlvirt{publishActiveEvents() : void}
      }

      \umlclass[x=10.5,y=0]{Event}{}{
        dispatch() : void
      }
      \umlclass[x=10.5,y=-2.5,type=abstract,fill=green!20]{EventReceiver}{}{
        publish() : void\\
        \umlvirt{dispatch() : void}
      }

      \umlclass[x=0.5,y=-5.3,type=abstract,fill=green!20]{TimerEventReceiver}{}{
        dispatch() : void\\
        \umlvirt{dispatchEvent() : void}\\
      }
      \umlclass[x=5.5,y=-5.3,type=abstract,fill=green!20]{DescriptorEventReceiver}{}{
        void enable(int fd)\\
        void disable()\\
        void suspend()\\
        void resume()\\
        dispatch() : void\\
        \umlvirt{dispatchEvent() : void}\\
        \umlvirt{timeoutEvent() : void}
      }

      \umlassoc{SNodeC}{EventLoop}
      \umlcompo[mult2=1]{EventLoop}{EventMultiplexer}
      \umlVHVcompo[mult2=3,pos2=2, weight=0.45]{EventMultiplexer}{DescriptorEventPublisher}
      \umlVHVcompo[mult2=1,pos2=2, weight=0.45]{EventMultiplexer}{TimerEventPublisher}
      \umlcompo[mult2=1]{EventMultiplexer}{EventQueue}
      \umluniassoc[mult2=*]{EventQueue}{Event}
      \umlcompo[mult1=1,mult2=1]{Event}{EventReceiver}
      \umluniassoc[mult2=*]{TimerEventPublisher}{TimerEventReceiver}
      \umluniassoc[mult2=*]{DescriptorEventPublisher}{DescriptorEventReceiver}
      \umlVHVinherit[anchor1=60, weight=0.55]{DescriptorEventReceiver}{EventReceiver}
      \umlVHVinherit[anchor1=60, weight=0.55]{TimerEventReceiver}{EventReceiver}
      %    \umlVHVcompo[weight=0.5]{epoll::EventMultiplexer}{EventMultiplexer}
    \end{tikzpicture}
  \end{center}
\end{frame}
