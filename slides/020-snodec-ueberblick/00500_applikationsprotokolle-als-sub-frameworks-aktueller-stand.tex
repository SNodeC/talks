\subsection{Application Protocols as Sub-Frameworks}

\begin{frame}{Application Protocols as Sub-Frameworks}{Current Status}
   \begin{description}[MQTT via WebSocket]
     \item [HTTP/0.9/1.0/1.1] Server and client side including \alert{virtual hosts} support.
     \item [HTTP/Express] Server side API equivalent to \alert{Node.JS/Express}
     \item [WebSocket] Server and client side
    \begin{itemize}
    \item \alert{Dynamically loaded} at runtime as \alert{plugin}.
    \item Subprotocols \alert{also loaded as plugins} at runtime.
    \item No recompile necessary in case a new Subprotocol is added.
    \end{itemize}
     \item [MQTT] Server and client side (\alert{full v3.1.1 support}, v5.0 in planning)
     \item [MQTT via WebSocket] Server and client side (what MQTT can do, MQTT via WS can automatically do too)
   \end{description}
\end{frame}
