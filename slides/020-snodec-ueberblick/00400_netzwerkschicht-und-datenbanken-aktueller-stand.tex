\subsection{Network Layer and Databases}

\begin{frame}{Network Layer and Databases}
   \begin{itemize}
     \item \texttt{SocketServer} and \texttt{SocketClients} classes (\texttt{ssl/tls} and \texttt{legacy} versions) for the network protocols
     \begin{itemize}
       \item \texttt{IPv4}
       \item \texttt{IPv6} (dual stack, single stack, IPv4 mapped addresses, \ldots)
       \item \texttt{rfcomm} (Bluetooth)
       \item \texttt{l2cap} (Bluetooth)
       \item \texttt{unix domain} Sockets
     \end{itemize}
     \item Database support
     \begin{itemize}
       \item \texttt{MariaDBClient} class (ready for use)
       \begin{itemize}
         \item "Normal"{} queries and execs
         \item transactions
       \end{itemize}
     \end{itemize}
   \end{itemize}
\end{frame}
