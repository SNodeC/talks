\subsection{WebSocket Application}

\begin{frame}[fragile]{WebSocket Server-Application}
  \begin{cppcode}[autogobble=true, caption={WebSocket Server-Application.},captionpos=b,morekeywords={[2]{upgrade}}]
    #include "express/legacy/WebApp.h"

    int main(int argc, char* argv[]) {
        core::SNodeC::init(argc, argv);

        express::legacy::WebApp webApp;

        webApp.get("/", [] APPLICATION(req, res) {
            res->upgrade(req, [req](bool success) -> void {
                if (success) {
                    std::cout << "Successful upgrade to '" << req->get("upgrade") << "'" << std::endl;
                } else {
                    std::cout << "Can not upgrade to '" << req->get("upgrade") << "'" << std::endl;
                }
            });
        });

        webApp.listen(8080, [](const SocketAddress& socketAddress,
                               const core::socket::State& state) -> void {
            // State processing
        });

        return core::SNodeC::start();
    }
  \end{cppcode}
\end{frame}
