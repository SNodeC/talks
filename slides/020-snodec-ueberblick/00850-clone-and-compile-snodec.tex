\section{Clone, Build, and Install the SNode.C Framework}
\withsectionpage

\begin{frame}{Clone, Build, and Install the SNode.C Framework}
  \begin{enumerate}
    \item Create a directory named \lstinline|projects| using the command line,
    \item change to that directory on the command line,
    \item create a directory named \lstinline|snode.c|,
    \item change to that directory.
    \item Clone SNode.C from \alert{\href{https://github.com/SNodeC/snode.c.git}{https://github.com/SNodeC/snode.c.git}} using \alert{HTTPS} as clone-method.
    \item Import the SNode.C framework into QT-Creator.
    \item Build and install SNode.C.
    \item Build the documentation by using QT-Creator.
  \end{enumerate}
\end{frame}
