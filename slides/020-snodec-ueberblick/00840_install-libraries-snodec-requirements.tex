\section{Install all Necessary Libraries for SNode.C and MQTTsuite}
\withsectionpage

\begin{frame}{Install all Necessary Libraries for SNode.C and MQTTsuite}
  \begin{itemize}
    \item Install all \alert{Hard Dependencies}:
    \begin{itemize}
      \item \texttt{\alert{libeasyloggingpp-dev}} for logging.
      \item \texttt{\alert{libssl-dev}} for SSL and TLS encrypted data exchange.
      \item \texttt{\alert{pkg-config}} package for cmake used to search for already installed libraries.
      \item \texttt{\alert{nlohmann-json3-dev}} for \alert{J}ava\alert{S}cript \alert{O}ptiont \alert{N}otation (\alert{JSON}) support.
    \end{itemize}
    \item Install all \alert{Soft Dependencies}:
    \begin{itemize}
      \item \texttt{\alert{clang-format}} to format the source code consistently.
      \item \texttt{\alert{cmake-format}} to format all CMakeLists.txt files consistently.
      \item \texttt{\alert{doxygen}} to create the documentation.
      \item \texttt{\alert{iwyu}} to prove that all required include files are explicitly included.
      \item \texttt{\alert{libbluetooth-dev}} for bluetooth transport.
      \item \texttt{\alert{libmagic-dev}} for mime-type-detection of files.
      \item \texttt{\alert{libmariadb-dev}} for database access.
    \end{itemize}
  \end{itemize}
\end{frame}
