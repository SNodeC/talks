\subsection{\texttt{SocketContextFactories} of the Server and Client}

\begin{frame}[fragile]{\texttt{SocketContextFactories} of the Server and Client}
  Server
  \begin{cppcode}[autogobble=true, caption={\alert{SocketContextFactory} of the server.},captionpos=b,morekeywords={[2]create},morekeywords={[3]{socketConnection}}]
    #include "EchoServerContext.h"
    #include <core/socket/stream/SocketContextFactory.h>

    class EchoServerContextFactory : public core::socket::stream::SocketContextFactory {
    private:
        core::socket::stream::SocketContext* create(core::socket::stream::SocketConnection* socketConnection) {
            return new EchoServerContext(socketConnection);
        }
    };
  \end{cppcode}
  Client
  \begin{cppcode}[autogobble=true, caption={\alert{SocketContextFactory} of the client.},captionpos=b,morekeywords={[2]create},morekeywords={[3]{socketConnection}}]
    #include "EchoClientContext.h"
    #include <core/socket/stream/SocketContextFactory.h>

    class EchoClientContextFactory : public core::socket::stream::SocketContextFactory {
    private:
        core::socket::stream::SocketContext* create(core::socket::stream::SocketConnection* socketConnection) {
            return new EchoClientContext(socketConnection);
        }
    };
  \end{cppcode}
\end{frame}
