\begin{frame}{Project Brief: Off-Grid Renewable Energy Solution}{for a Rural South African Community}
\begin{itemize}
    \item \textbf{Community:} around 250 households, rural and not connected to the public electricity grid.
    \item \textbf{Energy context:} peak demand of about 60\,kW and around 5.5 peak-sun-hours per day.
    \item \textbf{Reliability:} continuous supply for critical loads such as vaccine and agricultural-product refrigeration.
    \item \textbf{Solution:} renewable generation and storage that is replicable, locally maintainable and suitable for rural conditions.
\end{itemize}

\begin{block}{Three Pillars of the Overall Project}
\begin{enumerate}
    \item Reliable renewable \textbf{energy supply}
    \item Tangible \textbf{improvement of community life}
    \item \textbf{Education and community involvement}
\end{enumerate}
\end{block}

\begin{alertblock}{Question for the IoT Module}
How can sensing, communication, data collection and intelligent control make such a community energy system more efficient and resilient?
\end{alertblock}

\centering\footnotesize All student-built energy hardware remains SELV / approximately 5--24\,V DC; no mains-voltage construction or switching.
\end{frame}
