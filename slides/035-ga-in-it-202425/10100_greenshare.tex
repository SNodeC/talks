\section[nosectionpage]{GreenShare -- Smart Renewable Energy Community}

\begin{frame}{GreenShare -- Project Vision}
\begin{itemize}
    \item Treat households and community facilities as participants in a \textbf{shared renewable-energy community}.
    \item Continuously observe where energy is \textbf{produced, consumed, stored and needed}.
    \item Detect when one part of the community has renewable surplus while another has demand.
    \item Use available renewable energy more effectively instead of wasting surplus generation.
    \item Let selected flexible loads react automatically to the current community energy situation.
\end{itemize}

\begin{block}{Project Goal}
Build a working miniature renewable-energy community in which several households or community facilities coordinate generation, consumption, storage and flexible loads.
\end{block}
\end{frame}
