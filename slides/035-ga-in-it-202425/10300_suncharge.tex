\section[withsectionpage]{SunCharge -- Renewable-Aware E-Bike / Device Charging Station}

\begin{frame}{SunCharge -- Project Vision}
\begin{itemize}
    \item Renewable energy can directly support useful community services such as charging phones, devices, batteries or light electric mobility.
    \item A charging station must balance \textbf{available solar generation, stored energy, current charging demand and battery protection}.
    \item IoT makes the charging process observable and controllable.
    \item Charging can react to changing generation and storage conditions instead of operating blindly.
\end{itemize}

\begin{block}{Project Goal}
Build a small autonomous solar-powered charging station that intelligently manages its available renewable energy.
\end{block}
\end{frame}
