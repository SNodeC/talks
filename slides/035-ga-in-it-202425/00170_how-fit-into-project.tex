\section[withsectionpage][Biological key aspects]{Biologische Schlüsselaspekte \& Relevanz für das Monitoring des Hartbeespoort-Damms}

\begin{frame}{Art \& Grundlagenbiologie}
\begin{itemize}
\item \textbf{Species:} Water hyacinth (\textit{Eichhornia crassipes} / often also \textit{Pontederia crassipes})
\item \textbf{Growth form:} Free-floating freshwater macrophyte forming dense surface mats
\item \textbf{Reproduction:}
    \begin{itemize}
\item Predominantly \textbf{vegetative} via stolons (rapid clonal expansion)
\item Also \textbf{sexual} reproduction via seeds (longer-term persistence)
    \end{itemize}
\item \textbf{Ecological effects of mats:}
    \begin{itemize}
\item Strong \textbf{shading} $\rightarrow$ reduced underwater light, altered primary production
\item Reduzierter \textbf{Gasaustausch} und \textbf{gelöster Sauerstoff} (Risiko für Hypoxie/Anoxie)
\item Habitatveränderung; Auswirkungen auf Biodiversität und Ökosystemfunktion
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Warum sie invasiv ist (allgemeine Invasionsbiologie)}
\begin{itemize}
\item \textbf{Außerhalb ihres natürlichen Verbreitungsgebiets eingeführt} (Südamerika) und als selbsterhaltende Population etabliert
\item \textbf{Hohe intrinsische Wachstumsrate} und effiziente vegetative Vermehrung
\item \textbf{Ökologische Entlastung:} weniger wirksame natürliche Gegenspieler (Herbivoren/Pathogene) als im Herkunftsgebiet
\item \textbf{Breite Toleranz} gegenüber Umweltbedingungen; gedeiht besonders in nährstoffreichen Gewässern
\item \textbf{Strong competitive dominance:} rapid surface coverage blocks resources (light/space/oxygen)
\end{itemize}
\end{frame}

\begin{frame}{Wie sie vermutlich eingeschleppt wurde (Südafrika / Hartbeespoort-Kontext)}
\begin{itemize}
\item \textbf{Primary pathway:} historical \textbf{ornamental introduction} (pond/aquatic garden plant trade)
\item \textbf{Secondary spread:}
    \begin{itemize}
\item Movement of plant fragments between water bodies (boats, fishing gear, water transfers)
\item Vernetzung von Flüssen/Reservoirs ermöglicht Abdrift flussabwärts und Etablierung in geschützten Buchten
    \end{itemize}
\item \textbf{Lokale Verstärkung in Stauseen:} Eutrophierung und stabile Oberflächenwasserbedingungen fördern den raschen Biomasseaufbau
\end{itemize}
\end{frame}

\begin{frame}{Treiber der Ausbreitung und Verteilung (was steuert die Matten?)}
\begin{itemize}
\item \textbf{Nährstoffverfügbarkeit (N und P):} zentraler Treiber für Wachstumsrate, Biomasse und Mattendicke
\item \textbf{Hydrodynamics:}
    \begin{itemize}
\item Low-flow / sheltered zones favor establishment (bays, shorelines, embayments)
\item Wind-driven drift aggregates mats; fragmentation enables new colonies
    \end{itemize}
\item \textbf{Klima und Saisonalität:} warme Temperaturen und hohe Einstrahlung beschleunigen das Wachstum
\item \textbf{Störungsregime:} Stürme, Hochwasser und Managementmaßnahmen können Biomasse fragmentieren und umverteilen
\item \textbf{Biotische Regulation:} Wirksamkeit von Herbivoren/Pathogenen (natürlich oder für biologische Bekämpfung eingebracht)
\end{itemize}
\end{frame}

\begin{frame}{Warum Wasserchemie wichtig ist (und was zu beobachten ist)}
\begin{itemize}
\item Wasserhyazinthen werden stark durch \textbf{Eutrophierung} (Nährstoffüberschuss) begünstigt
\item \textbf{Nährstoffeinträge} (Abwässer, landwirtschaftlicher Oberflächenabfluss) entscheiden häufig darüber, ob Bekämpfung nachhaltig ist
\item \textbf{Key water-chemistry / physico-chemical variables:}
    \begin{itemize}
\item \textbf{Nährstoffe:} Nitrat, Ammonium, Phosphat (oft am aussagekräftigsten)
\item \textbf{Leitfähigkeit / TDS:} Proxy für Ionenfracht; kann mit Zuflüssen und Belastung korrelieren
\item \textbf{pH:} beeinflusst Nährstoffverfügbarkeit und biologischen Stress
\item \textbf{Temperatur:} steuert Stoffwechsel- und Wachstumsraten
\item \textbf{Trübung / Secchi-Tiefe:} Lichtdurchdringung; Wechselwirkung mit Beschattung und Algen
\item \textbf{Dissolved oxygen:} ecosystem health indicator; declines under dense mats
    \end{itemize}
\item \textbf{Interpretation:} chemistry $\rightarrow$ growth potential; mats $\rightarrow$ feedback on oxygen/light
\end{itemize}
\end{frame}

\begin{frame}{Mögliche Bekämpfung (integriertes Management)}
\begin{itemize}
\item \textbf{Quellenkontrolle (langfristig am wichtigsten):} Nährstoffeinträge reduzieren (Abwässer, Abfluss)
\item \textbf{Biologische Bekämpfung:}
    \begin{itemize}
\item Eingebrachte, spezialisierte Herbivoren (z.\,B. Rüsselkäfer) reduzieren Vitalität und Mattenbildung
\item Wirkt am besten als dauerhafter Druck; die Wirksamkeit hängt von Klima und Nährstoffstatus ab
    \end{itemize}
\item \textbf{Mechanical removal/harvesting:}
    \begin{itemize}
\item Rasche lokale Entlastung; hohe Betriebskosten; Risiko der Fragmentierung bei unsachgemäßer Durchführung
\item Erfordert eine Strategie zur Handhabung/Entsorgung (oder Nutzung) der Biomasse
    \end{itemize}
\item \textbf{Chemical control (herbicides):} kann wirksam sein, erfordert jedoch sorgfältige Regulierung, Timing und Folgenabschätzung
\item \textbf{Good Practice:} Methoden kombinieren + kontinuierliches Monitoring (Biologie + Chemie + Fernerkundung)
\end{itemize}
\end{frame}

\begin{frame}{Kernaussage für das Monitoring des Hartbeespoort-Damms}
\begin{itemize}
\item Die Dynamik der Wasserhyazinthen ist stark mit \textbf{nährstoffgetriebenem Wachstumspotenzial} verknüpft
\item \textbf{Real-time water chemistry} + surface mat tracking enables:
    \begin{itemize}
\item Early warning of high-growth conditions
\item Evaluation of interventions (nutrient reduction / removal / biocontrol)
\item Räumliches Verständnis: wo Matten akkumulieren und warum (Wind, Buchten, Zuflüsse)
    \end{itemize}
\item Nachhaltige Kontrolle erfordert \textbf{die Reduktion von Nährstoffeinträgen} zusätzlich zur direkten Biomassereduktion
\end{itemize}
\end{frame}
