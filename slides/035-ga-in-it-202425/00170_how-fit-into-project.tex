\subsection[Biological key aspects]{Biological key aspects \& relevance for Hartbeespoort Dam monitoring}

\begin{frame}{Species \& Core Biology}
\begin{itemize}
  \item \textbf{Species:} Water hyacinth (\textit{Eichhornia crassipes} / often also \textit{Pontederia crassipes})
  \item \textbf{Growth form:} Free-floating freshwater macrophyte forming dense surface mats
  \item \textbf{Reproduction:}
    \begin{itemize}
      \item Predominantly \textbf{vegetative} via stolons (rapid clonal expansion)
      \item Also \textbf{sexual} reproduction via seeds (longer-term persistence)
    \end{itemize}
  \item \textbf{Ecological effects of mats:}
    \begin{itemize}
      \item Strong \textbf{shading} $\rightarrow$ reduced underwater light, altered primary production
      \item Reduced \textbf{gas exchange} and \textbf{dissolved oxygen} (hypoxia/anoxia risk)
      \item Habitat transformation, impacts on biodiversity and ecosystem functioning
    \end{itemize}
\end{itemize}
\end{frame}

\begin{frame}{Why it is Invasive (general invasion biology)}
\begin{itemize}
  \item \textbf{Introduced outside its native range} (South America) and established self-sustaining populations
  \item \textbf{High intrinsic growth rate} and efficient vegetative reproduction
  \item \textbf{Ecological release:} fewer effective natural enemies (herbivores/pathogens) than in the native range
  \item \textbf{Broad tolerance} to environmental conditions; particularly thrives in nutrient-rich waters
  \item \textbf{Strong competitive dominance:} rapid surface coverage blocks resources (light/space/oxygen)
\end{itemize}
\end{frame}

\begin{frame}{How it Likely Arrived (South Africa / Hartbeespoort context)}
\begin{itemize}
  \item \textbf{Primary pathway:} historical \textbf{ornamental introduction} (pond/aquatic garden plant trade)
  \item \textbf{Secondary spread:}
    \begin{itemize}
      \item Movement of plant fragments between water bodies (boats, fishing gear, water transfers)
      \item Connectivity of rivers/reservoirs enabling downstream drift and establishment in sheltered bays
    \end{itemize}
  \item \textbf{Local amplification at reservoirs:} eutrophication and stable surface waters support rapid biomass build-up
\end{itemize}
\end{frame}

\begin{frame}{Drivers of Spread and Distribution (what controls the mats?)}
\begin{itemize}
  \item \textbf{Nutrient availability (N and P):} key driver of growth rate, biomass, and mat thickness
  \item \textbf{Hydrodynamics:}
    \begin{itemize}
      \item Low-flow / sheltered zones favor establishment (bays, shorelines, embayments)
      \item Wind-driven drift aggregates mats; fragmentation enables new colonies
    \end{itemize}
  \item \textbf{Climate and seasonality:} warm temperatures and high irradiance accelerate growth
  \item \textbf{Disturbance regime:} storms, floods, management actions can fragment and redistribute biomass
  \item \textbf{Biotic regulation:} effectiveness of herbivores/pathogens (natural or introduced for biocontrol)
\end{itemize}
\end{frame}

\begin{frame}{Why water chemistry matters (and what to measure)}
\begin{itemize}
  \item Water hyacinth is strongly promoted by \textbf{eutrophication} (excess nutrients)
  \item \textbf{Nutrient inputs} (wastewater, agricultural runoff) often determine whether control is sustainable
  \item \textbf{Key water-chemistry / physico-chemical variables:}
    \begin{itemize}
      \item \textbf{Nutrients:} nitrate, ammonium, phosphate (often the most predictive)
      \item \textbf{Conductivity / TDS:} proxy for ionic load; can correlate with inflows and pollution
      \item \textbf{pH:} influences nutrient availability and biological stress
      \item \textbf{Temperature:} controls metabolic and growth rates
      \item \textbf{Turbidity / Secchi depth:} light penetration; interacts with mat shading and algae
      \item \textbf{Dissolved oxygen:} ecosystem health indicator; declines under dense mats
    \end{itemize}
  \item \textbf{Interpretation:} chemistry $\rightarrow$ growth potential; mats $\rightarrow$ feedback on oxygen/light
\end{itemize}
\end{frame}

\begin{frame}{Possible control (integrated management)}
\begin{itemize}
  \item \textbf{Source control (most important long-term):} reduce nutrient loading (wastewater, runoff)
  \item \textbf{Biological control:}
    \begin{itemize}
      \item Introduced specialist herbivores (e.g., weevils) reduce vigor and mat formation
      \item Works best as sustained pressure; effectiveness depends on climate and nutrient status
    \end{itemize}
  \item \textbf{Mechanical removal/harvesting:}
    \begin{itemize}
      \item Rapid local relief; high operational cost; risk of fragmentation if poorly executed
      \item Requires biomass handling/disposal (or utilization) strategy
    \end{itemize}
  \item \textbf{Chemical control (herbicides):} can be effective but needs careful regulation, timing, and impact assessment
  \item \textbf{Best practice:} combine methods + continuous monitoring (biology + chemistry + remote sensing)
\end{itemize}
\end{frame}

\begin{frame}{Take-away for Hartbeespoort Dam monitoring}
\begin{itemize}
  \item Water hyacinth dynamics are strongly linked to \textbf{nutrient-driven growth potential}
  \item \textbf{Real-time water chemistry} + surface mat tracking enables:
    \begin{itemize}
      \item Early warning of high-growth conditions
      \item Evaluation of interventions (nutrient reduction / removal / biocontrol)
      \item Spatial understanding: where mats accumulate and why (wind, bays, inflows)
    \end{itemize}
  \item Sustainable control requires \textbf{addressing nutrient inputs} alongside direct biomass reduction
\end{itemize}
\end{frame}
