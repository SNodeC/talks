\begin{frame}{Wie können \alert{IoT und LoRaWAN} die Forschung der Stakeholder unterstützen?}{Die Grundidee}
\begin{description}[Messung der Wasserchemie]
\item[Messung der Wasserchemie]
\begin{itemize}
\item Platzieren Sie mehrere LoRaWAN-Gateways rund um den Victoriasee, ausgestattet mit
    \begin{itemize}
\item einem GPS-Empfänger zur Bestimmung der exakten Position sowie
\item Solarzellen und Akkumulatoren für einen autonomen Betrieb,
    \end{itemize}
    und integrieren Sie diese in \emph{The Things Network}.
\item Entwicklung eines schwimmenden LoRaWAN-Nodes, ausgestattet mit
    \begin{itemize}
\item einem GPS-Empfänger zur Bestimmung der exakten Position sowie
\item Solarzellen und Akkumulatoren für einen autonomen Betrieb,
    \end{itemize}
    und zur Messung von
    \begin{itemize}
\item Wasserqualitätsparametern,
\item der Nährstoffkonzentration im Wasser,
\item \ldots
    \end{itemize}
\item Integration des Nodes in \emph{The Things Network} sowie Übertragung der Messdaten und der GPS-Position an das Gateway (und damit auch an \emph{The Things Network}).
\item Entwicklung einer Weboberfläche für Management und Datenerfassung.
\end{itemize}
\end{description}
\end{frame}
