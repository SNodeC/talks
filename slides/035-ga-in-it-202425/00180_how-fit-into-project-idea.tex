\subsection{Idea for a Water Monitoring System}

\begin{frame}{How can \alert{IoT and LoRaWAN} support the Stakeholders' Research?}{The Big Idea}
\begin{description}[Measuring Water Chemistry]
\item[Measuring Water Chemistry] \begin{itemize}
    \item Place a couple of LoRaWAN-Gateways around the Victoria lake equipped with a
    \begin{itemize}
        \item GPS-Receiver to get the exact location of it and
        \item solar cells and accumulators to make the gateway autonomous, and
    \end{itemize}
    integrate them into The Things Network.
    \item Development of a swimming LoRaWAN-Node equipped with a
    \begin{itemize}
        \item GPS-Receiver to get the exact location of it and
        \item solar cells and accumulators to make the node autonomous, and
    \end{itemize}
    measuring
    \begin{itemize}
        \item water quality
        \item amount of nutrients in the water
        \item \ldots,
    \end{itemize}
    integrating it into The Things Network and sending the measured data and the GPS location to the gateway and therefore also to The Things Network.
    \item Development of a management and data acquiring Web interface.
\end{itemize}
\end{description}
\end{frame}
