\begin{frame}{GreenShare -- Prototype Scope \& Required Resources}
\small
\begin{columns}[T,onlytextwidth]
    \begin{column}{0.52\textwidth}
        \textbf{Hardware}
        \begin{itemize}
            \item Several energy nodes for households/community facilities
            \item ESP32-class controllers with LoRaWAN
            \item Voltage/current/power sensing, e.g. INA219/INA226
            \item Small PV modules or controllable low-voltage generation
            \item Representative loads and MOSFET/relay switching
            \item Optional small battery/storage nodes
        \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
        \textbf{Software \& Infrastructure}
        \begin{itemize}
            \item LoRaWAN gateway and The Things Network
            \item MQTT / MQTTSuite backend
            \item Database for current and historical data
            \item Energy-management logic
            \item Web dashboard
        \end{itemize}
    \end{column}
\end{columns}

\begin{block}{Final Demonstration}
Renewable surplus at one or more nodes is detected and results in a meaningful change in community energy use.
\end{block}
