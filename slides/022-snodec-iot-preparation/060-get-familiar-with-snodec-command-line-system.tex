\section{Learn Command Line Argument System of SNode.C Applications using -{}-help-all}

\begin{frame}{Learn Command Line Argument System of SNode.C Applications using -{}-help-all}
  \begin{enumerate}
    \item Start the SNode.C demo application \textcolor{fh-ooe-red}{\texttt{httpserver-legacy-in}}, a web server communicating unencrypted (\textcolor{fh-ooe-red}{legacy}) and accessible via IPv4 (\textcolor{fh-ooe-red}{in}) for static html projects, using the command line option \textcolor{fh-ooe-red}{\texttt{-{}-help-all}}.
    \item Analyse the help output.
    \item Start the \textcolor{fh-ooe-red}{\texttt{httpserver-legacy-in}} serving the SNode.C documentation and listening (bound) on \textcolor{fh-ooe-red}{port 8080}.
    \item Open an web browser and visit \texttt{\textcolor{fh-ooe-red}{\href{http://localhost:8080/}{http://localhost:8080/}}}.
    \item \textcolor{fh-ooe-red}{Application specific config-, log-, and pid-files} (in case application has been started as daemon) can be found under the directories \texttt{\textcolor{fh-ooe-red}{$\sim$/etc/snode.c/[conf, log, pid]/}}
  \end{enumerate}
\end{frame}
