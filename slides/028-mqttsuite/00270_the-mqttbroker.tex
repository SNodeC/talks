\part[nopartpage]{Using the MQTTsuite}

\section[withsectionpage]{The MQTTbroker}

\begin{frame}[fragile]{The MQTTbroker}
\begin{itemize}
\item It is prepared to accept connections via
    \begin{description}[xxxxxwss-mqtt]
        \item [mqtt] via IPv4 with standard port \alert{\lstinline|1883|}
        \item [mqtts] via IPv4 with standard port \alert{\lstinline|8883|}
        \item [mqtt] via unix domain socket with sun-path \alert{\lstinline|/tmp/mqttbroker|}
        \item [ws-mqtt] via IPv4 with port \alert{\lstinline|8080|} and path \alert{\lstinline|/ws/|}
        \item [wss-mqtt] via IPv4 with port \alert{\lstinline|8088|} and path \alert{\lstinline|/ws/|}
    \end{description}
\item The corresponding instances are
\begin{scriptsize}
\begin{Verbatim}[commandchars=\\\{\}]
\textcolor{grassgreen}{command@line~>} mqttbroker --help
\end{Verbatim}
\begin{Verbatim}[commandchars=\\\{\},frame=leftline,rulecolor=\color{fh-ooe-red}]
\ldots{} \ldots{} \ldots{} \ldots{} \ldots{} \ldots{}

Instances:
  \textcolor{fh-ooe-red}{in-mqtt}
       Configuration for server instance 'in-mqtt'
  \textcolor{fh-ooe-red}{in-mqtts}
       Configuration for server instance 'in-mqtts'
  \textcolor{fh-ooe-red}{un-mqtt}
       Configuration for server instance 'un-mqtt'
  \textcolor{fh-ooe-red}{in-http}
       Configuration for server instance 'in-http'
  \textcolor{fh-ooe-red}{in-https}
       Configuration for server instance 'in-https'
\end{Verbatim}
\end{scriptsize}
\end{itemize}
\end{frame}
