\begin{frame}{Clone, Build, and Install}
  \begin{enumerate}
    \item Under the directory \lstinline|projects| create a directory named \lstinline|mqttsuite|,
    \item change to that directory.
    \item Clone the MQTTSuite project from \alert{\href{https://github.com/SNodeC/mqttsuite.git}{https://github.com/SNodeC/mqttsuite.git}} using \alert{HTTPS} as clone-method.
    \item Import the MQTTSuite project into QT-Creator.
    \item Build and install MQTTSuite.
    \item Start the \texttt{\alert{mqttbroker}} and the \texttt{\alert{mqttintegrator}} applications separat in two different terminal windows.
    \item Watch the output of both applications.
    \item Use \texttt{\alert{mosquitto\textunderscore{}sub}} and  \texttt{\alert{mosquitto\textunderscore{}pub}} connected to the \texttt{\alert{mqttbroker}}.
    \item Try to find out what is going on in behind by interpreting the output.
  \end{enumerate}
\end{frame}
